240    def handle_area_calculation(self, checked: bool):
241        """Calcule et affiche ou masque l'aire des régions de la segmentation Watershed."""
242        try:
243            if not checked:
244                if hasattr(self.view, "watershed_area_label"):
245                    self.view.watershed_area_label.hide()
246                    self.view._update_image_info_position()
247                return
248
249            model = self.main_controller.model
250            # Vérifier si le watershed a été appliqué avant
251            if not hasattr(model, "watershed_labels") or model.watershed_labels is None:
252                self.error_controller.show_error("Erreur", "Veuillez appliquer la segmentation Watershed avant de calculer les aires.")
253                self.view.left_toolbar.btn_area.setChecked(False)
254                return
255
256            import numpy as np
257            import cv2
258
259            labels = model.watershed_labels
260            unique_labels = np.unique(labels)
261            unique_labels = unique_labels[unique_labels != 0]
262
263            if len(unique_labels) == 0:
264                self.error_controller.show_error("Erreur", "Aucune zone d'intérêt détectée par le Watershed.")
265                return
266
267            # Régénérer l'image couleur du watershed pour récupérer les couleurs exactes
268            image_affichage = cv2.normalize(labels, None, 0, 255, cv2.NORM_MINMAX, dtype=cv2.CV_8U)
269            image_couleur = cv2.applyColorMap(image_affichage, cv2.COLORMAP_JET)
270
271            html_lines = ["<html><body><p style='margin-top: 0px; margin-bottom: 6px; font-weight: bold;'>Aires du Watershed :</p>"]
272            html_lines.append("<table border='0' cellspacing='0' cellpadding='2'>")
273            for label in unique_labels:
274                # Récupérer la couleur de ce label
275                coords = np.argwhere(labels == label)
276                if len(coords) > 0:
277                    y, x = coords[0]
278                    b, g, r = image_couleur[y, x]
279                    color_hex = f"#{r:02x}{g:02x}{b:02x}"
280                else:
281                    color_hex = "#ffffff"
282
283                area_px = np.sum(labels == label)
284                # Utilisation d'un tableau HTML (pleinement supporté par le moteur de texte riche de Qt)
285                html_lines.append(
286                    f"<tr>"
287                    f"<td bgcolor='{color_hex}' width='12' height='12' style='border: 1px solid #555555;'>&nbsp;</td>"
288                    f"<td style='font-size: 11px; color: #e0e0e0; vertical-align: middle;'>&nbsp;&nbsp;{area_px} px²</td>"
289                    f"</tr>"
290                )
291            html_lines.append("</table></body></html>")
292            text = "".join(html_lines)
293
294            # Afficher dans la zone de même largeur
295            self.view.display_watershed_areas(text)
296
297        except Exception as e:
298            self.error_controller.handle_exception(e)
