 65    def handle_upload_et_sauvegarder(self, modalite: str | None = None) -> None:
 66        """
 67        Ouvre l'explorateur, affiche l'image dans la View,
 68        l'envoie sur le serveur via l'API
 69        ET sauvegarde le chemin serveur en BDD lié au patient courant.
 70        """
 71        # Guard hors-ligne
 72        if not is_online():
 73            self.error_handler.show_error("Mode hors-ligne", _MSG_OFFLINE)
 74            return None
 75
 76        try:
 77            # Vérifie qu'un patient est bien sélectionné avant d'uploader
 78            if self._current_patient_id is None:
 79                self.error_handler.show_error("Aucun patient sélectionné", "Sélectionnez ou créez un patient avant d'importer une image.")
 80                return
 81
 82            # Ouverture de l'explorateur de fichiers natif
 83            file_path, _ = QFileDialog.getOpenFileName(
 84                self.view,
 85                "Sélectionner une radiographie",
 86                "",
 87                "Images (*.png *.jpg *.jpeg *.dcm)",
 88            )
 89
 90            if not file_path:
 91                return  # L'utilisateur a annulé
 92
 93            # Enregistre le chemin du fichier chargé
 94            self.main_controller._last_file_path = file_path
 95
 96            # Conversion en numpy array normalisé [0,1] pour les traitements ultérieurs
 97            self.main_controller._current_array = ImageConvertie(file_path).convertirEnNumpyArray()
 98
 99            # Pour les DICOM, afficher via numpy array plutôt que QPixmap directement
100            if file_path.lower().endswith(".dcm"):
101                self.main_controller._display_numpy_array(self.main_controller._current_array)
102            else:
103                # Pour PNG/JPG, afficher directement et mémoriser
104                self.view.display_medical_image(file_path)
105
106            # Mémoriser le pixmap d'origine chargé depuis le fichier
107            if getattr(self.view, "current_pixmap", None) is not None:
108                self.main_controller._original_pixmap = self.view.current_pixmap.copy()
109
110            nom_fichier = os.path.basename(file_path)
111
112            # Envoi du fichier sur le serveur via l'API
113            # -> retourne le chemin distant ex: /home/ubuntu/esante/images/radio.png
114            chemin_serveur = upload_image(file_path, self._current_patient_id)
115
116            if chemin_serveur is None:
117                self.error_handler.show_error("Erreur serveur", "Impossible d'envoyer l'image au serveur.")
118                return
119
120            # Garde-fou parce que sauvegarder_image() attend un int et on envoie un int | None
121            if self._current_patient_id is None:
122                return
123
124            # Sauvegarde en BDD avec le chemin SERVEUR (pas le chemin local)
125            image_id = sauvegarder_image(
126                patient_id=self._current_patient_id,
127                nom_fichier=nom_fichier,
128                chemin=chemin_serveur,
129                modalite=modalite,
130            )
131            print(f"[ImageController] Image '{nom_fichier}' envoyée sur serveur et sauvegardée -> id={image_id}")
132
133        except Exception as e:
134            self.error_handler.handle_exception(e)

163    def handle_ouvrir_image_bdd(self, chemin: str) -> None:
164        """
165        Télécharge l'image depuis le serveur via l'API dans un fichier temporaire
166        et l'affiche dans la View.
167        Le chemin en BDD est un chemin serveur (ex: /home/ubuntu/esante/images/radio.png).
168        """
169        try:
170            # Extension du fichier pour que Qt l'affiche correctement
171            extension = os.path.splitext(chemin)[1] or ".png"
172
173            # Téléchargement dans un fichier temporaire local
174            # delete=False pour que le fichier survive après le with (Qt en a besoin)
175            with tempfile.NamedTemporaryFile(suffix=extension, delete=False) as tmp:
176                tmp_path = tmp.name
177
178            ok = download_image(chemin, tmp_path)
179
180            if not ok:
181                self.error_handler.show_error("Erreur serveur", f"Impossible de télécharger l'image depuis le serveur.")
182                return
183
184            # Enregistre le chemin du fichier chargé
185            self.main_controller._last_file_path = tmp_path
186
187            # Conversion en numpy array normalisé [0,1] pour les traitements ultérieurs
188            self.main_controller._current_array = ImageConvertie(tmp_path).convertirEnNumpyArray()
189
190            # Pour les DICOM, afficher via numpy array plutôt que QPixmap directement
191            if tmp_path.lower().endswith(".dcm"):
192                self.main_controller._display_numpy_array(self.main_controller._current_array)
193            else:
194                # Pour PNG/JPG, afficher directement
195                self.view.display_medical_image(tmp_path)
196
197            # Mémoriser le pixmap d'origine chargé depuis le fichier
198            if getattr(self.view, "current_pixmap", None) is not None:
199                self.main_controller._original_pixmap = self.view.current_pixmap.copy()
200
201            # RESET le slider de contraste à 1 pour repartir proprement
202            if self.view.contrast_slider is not None:
203                self.view.contrast_slider.setValue(1)
204
205            # Désactive les modes actifs pour éviter les états parasites
206            if self.view.contrast_slider_active:
207                self.view.toggle_contrast_slider_mode(False)
208                self.view.left_toolbar.btn_contrast_slider.setChecked(False)
209
210            print(f"[ImageController] Image ouverte depuis serveur : {chemin}")
211
212        except Exception as e:
213            self.error_handler.handle_exception(e)

244    def handle_export_image(self) -> None:
245        """
246        Exporte l'image affichée à l'écran avec ses annotations, règles et formes superposées.
247        Enregistre le résultat sous forme d'image (PNG, JPEG) ou document PDF.
248        """
249        try:
250            # Vérifier qu'une image est bien chargée et affichée
251            if self.main_controller._original_pixmap is None or self.view.current_pixmap is None:
252                self.error_handler.show_error("Aucune image chargée", "Il n'y a aucune image à exporter.")
253                return
254
255            pixmap_displayed = self.view.image_display.pixmap()
256            if not pixmap_displayed or pixmap_displayed.isNull():
257                self.error_handler.show_error("Aucune image affichée", "Il n'y a pas d'image affichée à exporter.")
258                return
259
260            # Nom par défaut suggéré
261            nom_defaut = "export_image.png"
262            last_path = self.main_controller._last_file_path
263            if last_path and last_path != "from_controller":
264                base_name = os.path.splitext(os.path.basename(last_path))[0]
265                nom_defaut = f"{base_name}_export.png"
266
267            # Ouvre le dialogue de sauvegarde avec PDF inclus
268            file_path, _ = QFileDialog.getSaveFileName(
269                self.view, "Exporter l'image avec annotations", nom_defaut, "Images PNG (*.png);;Images JPEG (*.jpg *.jpeg);;Documents PDF (*.pdf);;Tous les fichiers (*)"
270            )
271
272            if not file_path:
273                return  # Annulé par l'utilisateur
274
275            # Capture le widget d'affichage (QLabel)
276            full_pixmap = self.view.image_display.grab()
277
278            # Calcul des marges pour ne rogner que la zone de l'image
279            margin_x = (self.view.image_display.width() - pixmap_displayed.width()) // 2
280            margin_y = (self.view.image_display.height() - pixmap_displayed.height()) // 2
281            img_rect = QRect(margin_x, margin_y, pixmap_displayed.width(), pixmap_displayed.height())
282
283            # Rogner pour garder l'image exacte
284            cropped_pixmap = full_pixmap.copy(img_rect)
285
286            success = False
287            # Enregistrement
288            if file_path.lower().endswith(".pdf"):
289                try:
290                    from PyQt6.QtGui import QPdfWriter, QPainter, QPageSize, QPageLayout
291
292                    writer = QPdfWriter(file_path)
293                    writer.setPageSize(QPageSize(QPageSize.PageSizeId.A4))
294
295                    if cropped_pixmap.width() > cropped_pixmap.height():
296                        writer.setPageOrientation(QPageLayout.Orientation.Landscape)
297                    else:
298                        writer.setPageOrientation(QPageLayout.Orientation.Portrait)
299
300                    painter = QPainter(writer)
301                    w = writer.width()
302                    h = writer.height()
303                    scaled = cropped_pixmap.scaled(w, h, Qt.AspectRatioMode.KeepAspectRatio, Qt.TransformationMode.SmoothTransformation)
304                    x = (w - scaled.width()) // 2
305                    y = (h - scaled.height()) // 2
306                    painter.drawPixmap(x, y, scaled)
307                    painter.end()
308                    success = True
309                except Exception as pdf_error:
310                    print(f"Erreur lors de la génération du PDF : {pdf_error}")
311                    success = False
312            else:
313                success = cropped_pixmap.save(file_path)
314
315            if success:
316                print(f"[ImageController] Image exportée avec succès : {file_path}")
317                # Demander si l'utilisateur veut aussi l'ajouter au dossier patient
318                from PyQt6.QtWidgets import QMessageBox
319
320                reply = QMessageBox.question(
321                    self.view,
322                    "Enregistrer dans le dossier patient",
323                    "L'image a été exportée localement.\nVoulez-vous également l'enregistrer dans le dossier médical de ce patient ?",
324                    QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No,
325                )
326                if reply == QMessageBox.StandardButton.Yes:
327                    self.handle_save_to_patient_record(cropped_pixmap)
328            else:
329                self.error_handler.show_error("Erreur d'exportation", "Impossible d'enregistrer l'image ou le document PDF. Vérifiez les permissions.")
330
331        except Exception as e:
332            self.error_handler.handle_exception(e)

338        """
339        Enregistre l'image annotée courante dans la base de données et sur le serveur

343        if not is_online():
344            self.error_handler.show_error("Mode hors-ligne", "Cette fonctionnalité nécessite une connexion Internet.")

347        try:
348            # Vérifier qu'un patient est sélectionné

350                self.error_handler.show_error("Aucun patient sélectionné", "Veuillez sélectionner un patient avant d'enregistrer l'image dans son dossier.")
351                return
352
353            # Si cropped_pixmap n'est pas fourni, on le capture à la volée
354            if cropped_pixmap is None:
355                if self.main_controller._original_pixmap is None or self.view.current_pixmap is None:
356                    self.error_handler.show_error("Aucune image chargée", "Il n'y a aucune image à enregistrer.")

364                # Capture le widget d'affichage (QLabel)
365                full_pixmap = self.view.image_display.grab()
366
367                # Calcul des marges pour ne rogner que la zone de l'image
368                margin_x = (self.view.image_display.width() - pixmap_displayed.width()) // 2
369                margin_y = (self.view.image_display.height() - pixmap_displayed.height()) // 2
370                img_rect = QRect(margin_x, margin_y, pixmap_displayed.width(), pixmap_displayed.height())
371
372                # Rogner pour garder l'image exacte
373                cropped_pixmap = full_pixmap.copy(img_rect)
374
375            # Demander le nom du fichier à l'utilisateur (ou générer un nom automatique)
376            last_path = self.main_controller._last_file_path
377            nom_defaut = "radio_annotee.png"
378            if last_path and last_path != "from_controller":
379                base_name = os.path.splitext(os.path.basename(last_path))[0]
380                nom_defaut = f"{base_name}_annotee.png"
381
382            # Boîte de dialogue simple pour saisir le nom de l'image dans le dossier
383            from PyQt6.QtWidgets import QInputDialog
384
385            nom_fichier, ok = QInputDialog.getText(self.view, "Enregistrer dans le dossier", "Nom de la radiographie annotée :", text=nom_defaut)
386            if not ok or not nom_fichier.strip():
387                return  # Annulé
388
389            # S'assurer de l'extension .png par défaut si aucune extension d'image valide n'est saisie
390            _, ext = os.path.splitext(nom_fichier)
391            if not ext.lower() in [".png", ".jpg", ".jpeg"]:
392                nom_fichier += ".png"
393
394            # Sauvegarde temporaire locale du pixmap pour l'upload
395            import tempfile
396
397            with tempfile.NamedTemporaryFile(suffix=".png", delete=False) as tmp:
398                tmp_path = tmp.name
399
400            cropped_pixmap.save(tmp_path)
401
402            # Upload et sauvegarde BDD
403            chemin_serveur = upload_image(tmp_path, self._current_patient_id)
404
405            # Supprimer le fichier temporaire local
406            try:
407                os.unlink(tmp_path)
408            except:
409                pass
410
411            if chemin_serveur is None:
412                self.error_handler.show_error("Erreur serveur", "Impossible d'envoyer l'image au serveur.")
413                return
414
415            image_id = sauvegarder_image(patient_id=self._current_patient_id, nom_fichier=nom_fichier, chemin=chemin_serveur, modalite="ANNOTATED")
416
417            print(f"[ImageController] Image annotée '{nom_fichier}' enregistrée en BDD (id={image_id})")
418
419            # Message de succès
420            self.error_handler.show_info("Enregistrement réussi", f"L'image annotée '{nom_fichier}' a été ajoutée avec succès au dossier du patient.")
421
422            # Rafraîchir la liste des images du patient dans la vue
423            p_manager = self.view.left_toolbar.patient_manager
424            images = self.handle_charger_images_patient(self._current_patient_id)
425            p_manager.refresh_images(images)
426
427        except Exception as e:
428            self.error_handler.handle_exception(e)

278
