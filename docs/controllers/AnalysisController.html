 72    def handle_tfd2d(self, checked: bool):
 73        """
 74        Applique la Transformée de Fourier Discrète 2D sur l'image courante
 75        et l'affiche/masque sous forme de vignette (overlay) au-dessus des infos de l'image.
 76        Nécessite qu'une image soit chargée (_current_array != None).
 77        """
 78        try:
 79            if self._current_array is None:
 80                self.error_handler.show_error("Erreur", "Aucune image chargée")
 81                self.view.top_toolbar.btn_fft.setChecked(False)
 82                return
 83
 84            if checked:
 85                # Désactiver l'histogramme si actif
 86                self.view.top_toolbar.btn_histo.setChecked(False)
 87                self.view.hide_histogram()
 88
 89                # Calcul du spectre fréquentiel via la TFD2D
 90                tfd2d = TFD2D(self._current_array)
 91                spectre = tfd2d.calculerTFDSpectre()
 92
 93                # Normalisation [0,1] — on évite la division par zéro si le spectre est vide
 94                max_val = spectre.max()
 95                norm_spectre = spectre / max_val if max_val > 0 else spectre
 96
 97                # Conversion en QPixmap
 98                img_uint8 = (norm_spectre * 255).astype(np.uint8)
 99                h, w = img_uint8.shape
100                from PyQt6.QtGui import QImage, QPixmap
101
102                qimage = QImage(bytes(img_uint8.data), w, h, w, QImage.Format.Format_Grayscale8).copy()
103                pixmap = QPixmap.fromImage(qimage)
104
105                self.view.display_fft_spectrum(pixmap)
106                print("Spectre FFT affiché dans l'overlay.")
107            else:
108                self.view.hide_fft_spectrum()
109                print("Spectre FFT masqué.")
110
111        except Exception as e:
112            self.view.top_toolbar.btn_fft.setChecked(False)
113            self.error_handler.handle_exception(e)
114            return

211    def apply_contrast_realtime(self, facteur_contraste):
212        """
213        Applique le contraste en temps réel à partir de l'image de base.
214        Cette méthode est appelée à chaque mouvement du slider.
215
216        :param facteur_contraste: Valeur du slider (1-10)
217        """
218        try:
219            if self._contrast_base_array is None:
220                return
221
222            # Application de l'ajustement de contraste sur l'image de base
223            # new_pixel = 0.5 + facteur * (old_pixel - 0.5)
224            result = 0.5 + facteur_contraste * (self._contrast_base_array - 0.5)
225            result = np.clip(result, 0, 1).astype(np.float32)
226
227            # Mettre à jour _current_array pour que l'affichage reflète les modifications
228            self._current_array = result
229
230            # Affichage
231            self._display_numpy_array(result)
232
233        except Exception as e:
234            self.error_handler.handle_exception(e)
235            return

260                result = (masque / 255.0).astype(np.float32)
261                self._display_numpy_array(result)
262
263                mode = f"Otsu (seuil calculé: {int(seuil_calcule)})" if seuil_manuel is None else f"Manuel: {seuil_manuel}"
264                print(f"Seuillage appliqué - {mode}")
265
266        except Exception as e:
267            self.error_handler.handle_exception(e)
268            return

270    def handle_watershed(self, checked):
271        """
272        Applique la segmentation Watershed sur l'image courante.
273        Nécessite qu'une image soit chargée (_current_array != None).
274        """
275        try:
276            if self._current_array is None:
277                self.error_handler.show_error("Erreur", "Aucune image chargée")
278                self.view.left_toolbar.btn_watershed.setChecked(False)
279                return
280            if checked:
281                self.main_controller.handle_reset_image(keep_button=self.view.left_toolbar.btn_watershed)
282                self.main_controller.model.watershed_labels = None
283                if hasattr(self.view, "watershed_area_label"):
284                    self.view.watershed_area_label.hide()
285                self.view.left_toolbar.btn_area.setChecked(False)
286                pipette_seuil = getattr(self.main_controller, "last_pipette_threshold", None)
287                if pipette_seuil is not None:
288                    seuil_otsu = False
289                    seuil = pipette_seuil
290                    self.main_controller.last_pipette_threshold = None
291                else:
292                    seuil_otsu = self.last_watershed_otsu
293                    seuil = self.last_watershed_seuil
294
295                dialog = WatershedDialog(
296                    self.view,
297                    default_sigma=self.last_watershed_sigma,
298                    default_seuil_otsu=seuil_otsu,
299                    default_seuil=seuil,
300                    default_kernel=self.last_watershed_kernel,
301                    default_min_dist=self.last_watershed_min_dist,
302                )
303                if dialog.exec():
304                    sigma, seuil_manuel, kernel_size, min_dist = dialog.get_values()
305                    self.last_watershed_sigma = sigma
306                    self.last_watershed_otsu = seuil_manuel is None
307                    self.last_watershed_seuil = seuil_manuel if seuil_manuel is not None else 40
308                    self.last_watershed_kernel = kernel_size
309                    self.last_watershed_min_dist = min_dist
310
311                    # 1. Filtrage Gaussien
312                    image_filtree = FiltrageGaussien(sigma, self._current_array).filtrage()
313
314                    # 2. Seuillage (manuel ou Otsu si seuil_manuel est None)
315                    outil_seuillage = Seuillage(seuil_manuel)
316                    masque, seuil_choisi = outil_seuillage.appliquer(image_filtree)
317
318                    # 3. Masque des cavités internes
319                    masque_rempli = ndimage.binary_fill_holes(masque > 0)
320                    masque_rempli = ((masque_rempli * 255).astype(np.uint8) > 0) & (masque == 0)
321                    masque_rempli = (masque_rempli * 255).astype(np.uint8)
322
323                    # 4. Nettoyage morphologique
324                    masque_propre = MorphologieMathematique(kernel_size).ouverture(masque_rempli)
325
326                    # 5. Segmentation Watershed
327                    labels = SegmentationWatershed(min_dist).segmenter(masque_propre)
328
329                    # 6. Rendu et superposition
330                    image_affichage = cv2.normalize(labels, None, 0, 255, cv2.NORM_MINMAX, dtype=cv2.CV_8U)
331                    image_couleur = cv2.applyColorMap(image_affichage, cv2.COLORMAP_JET)
332
333                    img_8bit = (self._current_array * 255).astype(np.uint8)
334                    img_bgr = cv2.cvtColor(img_8bit, cv2.COLOR_GRAY2BGR)
335
336                    # superposition transparente (overlay) sur les zones segmentées
337                    mask_roi = labels > 0
338                    overlay = img_bgr.copy()
339                    blended = cv2.addWeighted(img_bgr, 0.6, image_couleur, 0.4, 0)
340                    overlay[mask_roi] = blended[mask_roi]
341
342                    self._display_numpy_array(overlay)
343                    self.main_controller.model.watershed_labels = labels
344
345                    print(f"Segmentation Watershed appliquée (sigma={sigma}, seuil={seuil_choisi}, noyau={kernel_size}, dist={min_dist})")
346                else:
347                    self.main_controller.handle_reset_image()
348            else:
349                self.main_controller.handle_reset_image()
350
351        except Exception as e:
352            self.view.left_toolbar.btn_watershed.setChecked(False)
353            self.error_handler.handle_exception(e)
354            return
