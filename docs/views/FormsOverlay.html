 77    def handle_mouse_press(self, local_pos: QPoint, img_rect: QRect) -> bool:
 78        """Gère le clic de la souris."""
 79        if not img_rect.contains(local_pos):
 80            return False
 81
 82        sx = local_pos.x()
 83        sy = local_pos.y()
 84        rx, ry = self.to_relative(sx, sy, img_rect)
 85
 86        # 1. Si une forme est déjà sélectionnée, vérifier si on clique sur l'une de ses poignées
 87        if self.selected_shape is not None:
 88            shape = self.selected_shape
 89            scx, scy = self.to_screen(shape.cx, shape.cy, img_rect)
 90            sw = shape.w * img_rect.width()
 91            sh = shape.h * img_rect.height()
 92            rad = math.radians(shape.rotation)
 93
 94            # Poignée de rotation (0, -sh/2 - 25)
 95            rx_rot, ry_rot = self.rotate_point(0, -sh / 2 - 25, rad)
 96            rot_handle_pos = (scx + rx_rot, scy + ry_rot)
 97            if self.dist((sx, sy), rot_handle_pos) < 10:
 98                self.interaction_mode = "rotating"
 99                self.initial_rotation = shape.rotation
100                self.initial_mouse_angle = math.degrees(math.atan2(sy - scy, sx - scx))
101                return True
102
103            # Poignées de redimensionnement aux coins
104            corners = [("tl", (-sw / 2, -sh / 2)), ("tr", (sw / 2, -sh / 2)), ("bl", (-sw / 2, sh / 2)), ("br", (sw / 2, sh / 2))]
105            for name, (lx, ly) in corners:
106                rx_c, ry_c = self.rotate_point(lx, ly, rad)
107                corner_pos = (scx + rx_c, scy + ry_c)
108                if self.dist((sx, sy), corner_pos) < 10:
109                    self.interaction_mode = "resizing"
110                    self.active_handle = name
111                    return True
112
113        # 2. Vérifier si on clique à l'intérieur d'une forme existante pour la sélectionner/déplacer
114        for shape in reversed(self.shapes):
115            scx, scy = self.to_screen(shape.cx, shape.cy, img_rect)
116            sw = shape.w * img_rect.width()
117            sh = shape.h * img_rect.height()
118            rad = math.radians(shape.rotation)
119
120            dx = sx - scx
121            dy = sy - scy
122            lx, ly = self.rotate_point(dx, dy, -rad)
123
124            inside = False
125            if shape.shape_type == "circle":
126                inside = (lx / (sw / 2)) ** 2 + (ly / (sh / 2)) ** 2 <= 1.0
127            else:
128                inside = abs(lx) <= sw / 2 and abs(ly) <= sh / 2
129
130            if inside:
131                if self.selected_shape:
132                    self.selected_shape.selected = False
133                self.selected_shape = shape
134                shape.selected = True
135                self.interaction_mode = "moving"
136                self.drag_offset_x = rx - shape.cx
137                self.drag_offset_y = ry - shape.cy
138                return True
139
140        # 3. Finaliser le tracé en cours s'il s'agit du deuxième clic de création
141        if self.interaction_mode == "creating":
142            self.rel_b = (rx, ry)
143            dx = abs(self.rel_b[0] - self.rel_a[0])
144            dy = abs(self.rel_b[1] - self.rel_a[1])
145            if dx > 0.005 or dy > 0.005:
146                if self.shape_type == "circle":
147                    cx = self.rel_a[0]
148                    cy = self.rel_a[1]
149                    r = math.sqrt((self.rel_b[0] - self.rel_a[0]) ** 2 + (self.rel_b[1] - self.rel_a[1]) ** 2)
150                    w = h = 2.0 * r
151                else:
152                    cx = (self.rel_a[0] + self.rel_b[0]) / 2.0
153                    cy = (self.rel_a[1] + self.rel_b[1]) / 2.0
154                    w = dx
155                    h = dy
156
157                new_shape = Shape(self.shape_type, cx, cy, w, h, 0.0)
158                new_shape.selected = True
159                if self.selected_shape:
160                    self.selected_shape.selected = False
161                self.selected_shape = new_shape
162                self.shapes.append(new_shape)
163
164            self.rel_a = None
165            self.rel_b = None
166            self.rel_temp = None
167            self.interaction_mode = None
168            return True
169
170        # 4. Clic dans le vide : désélectionner et initier le tracé d'une nouvelle forme
171        if self.selected_shape:
172            self.selected_shape.selected = False
173            self.selected_shape = None
174
175        self.interaction_mode = "creating"
176        self.rel_a = (rx, ry)
177        self.rel_b = None
178        self.rel_temp = None
179        return True

181    def handle_mouse_move(self, local_pos: QPoint, img_rect: QRect) -> bool:
182        """Gère le déplacement de la souris."""
183        sx = local_pos.x()
184        sy = local_pos.y()
185
186        img_x = max(img_rect.left(), min(sx, img_rect.right()))
187        img_y = max(img_rect.top(), min(sy, img_rect.bottom()))
188        rx, ry = self.to_relative(img_x, img_y, img_rect)
189
190        if self.interaction_mode == "creating":
191            self.rel_temp = (rx, ry)
192            return True
193
194        elif self.interaction_mode == "moving" and self.selected_shape:
195            shape = self.selected_shape
196            shape.cx = rx - self.drag_offset_x
197            shape.cy = ry - self.drag_offset_y
198            shape.cx = max(0.0, min(shape.cx, 1.0))
199            shape.cy = max(0.0, min(shape.cy, 1.0))
200            return True
201
202        elif self.interaction_mode == "rotating" and self.selected_shape:
203            shape = self.selected_shape
204            scx, scy = self.to_screen(shape.cx, shape.cy, img_rect)
205            current_angle = math.degrees(math.atan2(sy - scy, sx - scx))
206            delta_angle = current_angle - self.initial_mouse_angle
207            shape.rotation = (self.initial_rotation + delta_angle) % 360.0
208            return True
209
210        elif self.interaction_mode == "resizing" and self.selected_shape:
211            shape = self.selected_shape
212            scx, scy = self.to_screen(shape.cx, shape.cy, img_rect)
213            rad = math.radians(shape.rotation)
214
215            dx = sx - scx
216            dy = sy - scy
217            lx, ly = self.rotate_point(dx, dy, -rad)
218
219            new_w = 2.0 * abs(lx) / img_rect.width()
220            new_h = 2.0 * abs(ly) / img_rect.height()
221
222            min_size = 0.005
223            new_w = max(min_size, new_w)
224            new_h = max(min_size, new_h)
225
226            if shape.shape_type == "circle":
227                dist_local = math.sqrt(lx**2 + ly**2)
228                new_w = 2.0 * dist_local / img_rect.width()
229                shape.w = max(min_size, new_w)
230                shape.h = shape.w * (img_rect.width() / img_rect.height())
231            else:
232                shape.w = new_w
233                shape.h = new_h
234
235            return True
236
237        else:
238            self.update_hover_cursor(sx, sy, img_rect)
239
240        return False

307        """Récupère l'image d'origine sous forme de tableau NumPy non normalisé avec métadonnées."""
308        main_view = self.label_view.visualizer.main_view
309        if not main_view:
310            return None, ""
311
312        file_path = main_view.current_file_path
313        if (not file_path or file_path == "from_controller") and main_view.controller:
314            file_path = getattr(main_view.controller, "_last_file_path", None)
315
316        if not file_path or not os.path.exists(file_path):
317            return None, ""
318
319        if self.cached_file_path != file_path:
320            try:
321                if file_path.lower().endswith(".dcm"):
322                    import pydicom
323
324                    dicom_image = pydicom.dcmread(file_path)
325                    pixel_array = dicom_image.pixel_array.astype(np.float32)
326                    slope = getattr(dicom_image, "RescaleSlope", 1.0)
327                    intercept = getattr(dicom_image, "RescaleIntercept", 0.0)
328                    self.cached_raw_pixels = pixel_array * slope + intercept
329                    self.cached_unit = "HU"
330                else:
331                    from PIL import Image
332
333                    img = Image.open(file_path).convert("L")
334                    self.cached_raw_pixels = np.array(img).astype(np.float32)
335                    self.cached_unit = ""
336                self.cached_file_path = file_path
337            except Exception as e:
338                print(f"Error loading raw pixels: {e}")
339                return None, ""
340
341        return self.cached_raw_pixels, self.cached_unit

453    def draw_shape_stats(self, painter: QPainter, img_rect: QRect, shape: Shape):
454        """Calcule les statistiques et dessine la boîte d'informations."""
455        raw_pixels, unit = self.get_raw_pixels()
456        if raw_pixels is None:
457            return
458
459        orig_h, orig_w = raw_pixels.shape
460        orig_cx = shape.cx * orig_w
461        orig_cy = shape.cy * orig_h
462        orig_rw = shape.w * orig_w
463        orig_rh = shape.h * orig_h
464
465        # Calculer le AABB (Axis-Aligned Bounding Box) pour limiter la zone de scan
466        orig_rad = math.radians(shape.rotation)
467        orig_corners = [
468            self.rotate_point(-orig_rw / 2, -orig_rh / 2, orig_rad),
469            self.rotate_point(orig_rw / 2, -orig_rh / 2, orig_rad),
470            self.rotate_point(-orig_rw / 2, orig_rh / 2, orig_rad),
471            self.rotate_point(orig_rw / 2, orig_rh / 2, orig_rad),
472        ]
473        orig_xs = [orig_cx + dx for dx, dy in orig_corners]
474        orig_ys = [orig_cy + dy for dx, dy in orig_corners]
475
476        xmin = max(0, int(math.floor(min(orig_xs))))
477        xmax = min(orig_w - 1, int(math.ceil(max(orig_xs))))
478        ymin = max(0, int(math.floor(min(orig_ys))))
479        ymax = min(orig_h - 1, int(math.ceil(max(orig_ys))))
480
481        # Extraction vectorisée NumPy avec rotation inverse pour le calcul du masque
482        if xmax >= xmin and ymax >= ymin:
483            Y, X = np.ogrid[ymin : ymax + 1, xmin : xmax + 1]
484            cos_a = math.cos(orig_rad)
485            sin_a = math.sin(orig_rad)
486
487            # Transformation inverse vers le repère local de la ROI
488            local_X = (X - orig_cx) * cos_a + (Y - orig_cy) * sin_a
489            local_Y = -(X - orig_cx) * sin_a + (Y - orig_cy) * cos_a
490
491            if shape.shape_type == "circle":
492                # Le cercle utilise orig_rw comme diamètre (équivalent à sw à l'échelle d'origine)
493                mask = (local_X**2 + local_Y**2) <= (orig_rw / 2.0) ** 2
494            else:
495                mask = (np.abs(local_X) <= orig_rw / 2.0) & (np.abs(local_Y) <= orig_rh / 2.0)
496
497            sub_pixels = raw_pixels[ymin : ymax + 1, xmin : xmax + 1]
498            values = sub_pixels[mask]
499        else:
500            values = np.array([])
501
502        if values.size > 0:
503            mean_val = np.mean(values)
504            max_val = np.max(values)
