156        self.lbl_left_val.setText(str(int(histo_model.left_val)))

170    def update_plot(self):
171        if self.plot_data is None:
172            return
173
174        # Calculer l'histogramme avec 256 bacs (bins) pour l'affichage
175        hist, bin_edges = np.histogram(self.plot_data, bins=256, range=self.plot_range)
176        bin_centers = (bin_edges[:-1] + bin_edges[1:]) / 2
177
178        dpi = 100
179        # 300x120 pixels en pouces pour matplotlib
180        fig, ax = plt.subplots(figsize=(300 / dpi, 120 / dpi), dpi=dpi, facecolor="#1a1a1a")
181        ax.set_facecolor("#1a1a1a")
182
183        # Tracé de la courbe et de son aire
184        ax.plot(bin_centers, hist, color="#00a2ed", linewidth=1.2)
185        ax.fill_between(bin_centers, hist, color="#00a2ed", alpha=0.25)
186
187        # Gestion de l'échelle linéaire / logarithmique
188        if not self.linear_scale:
189            ax.set_yscale("log")
190            # Fixer une valeur minimale pour éviter un rendu brisé si certains bacs sont vides
191            ax.set_ylim(bottom=0.8)
192
193        # Grille verticale uniquement (comme sur la capture utilisateur)
194        ax.grid(True, which="both", axis="x", color="#3c3c3c", linestyle="-", linewidth=0.5, alpha=0.5)
195        ax.grid(False, axis="y")
196
197        # Masquer les bordures inutiles
198        ax.spines["top"].set_visible(False)
199        ax.spines["right"].set_visible(False)
200        ax.spines["left"].set_visible(False)
201        ax.spines["bottom"].set_color("#3c3c3c")
202        ax.spines["bottom"].set_linewidth(0.5)
203
204        # Masquer les axes et graduations Y, n'afficher que des graduations X minimales
205        ax.yaxis.set_visible(False)
206        ax.tick_params(colors="#888888", labelsize=8, length=0)
207        ax.set_xticklabels([])  # Pas de texte de graduations (géré par l'en-tête du widget)
208
209        # Ajuster les espacements
210        plt.tight_layout()
211
212        # Exporter l'image au format PNG en mémoire
213        buf = io.BytesIO()
214        plt.savefig(buf, format="png", facecolor=fig.get_facecolor(), edgecolor="none", bbox_inches="tight")
215        plt.close(fig)
216        buf.seek(0)
217
218        # Charger dans un QPixmap et l'afficher
219        pixmap = QPixmap()
220        pixmap.loadFromData(buf.read())
221        self.plot_label.setPixmap(pixmap)

224        self.linear_scale = self.chk_linear.isChecked()
