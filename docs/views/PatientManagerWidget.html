 81    def on_patient_selected(self):
 82        selected_items = self.patient_list.selectedItems()
 83        if not selected_items:
 84            if self.main_view.controller:  # Si le controller renvoie None, permet d'eviter le crash
 85                self.main_view.controller._current_patient_id = None
 86            self.image_list.clear()
 87            # Masquer le volet si plus rien n'est sélectionné
 88            self.main_view.patient_info_panel.display_patient(None)
 89            return
 90
 91        item = selected_items[0]
 92        patient_id = item.data(Qt.ItemDataRole.UserRole)
 93        patient_tuple = item.data(Qt.ItemDataRole.UserRole + 1)  # ← récupère le tuple en mémoire
 94
 95        self.main_view.controller._current_patient_id = patient_id
 96
 97        if self.main_view.controller:
 98
 99            # 2. On transmet le tuple trouvé au panneau d'affichage à droite de la barre d'outils
100            self.main_view.patient_info_panel.display_patient(patient_tuple)
101
102            # 3. On charge les images normalement
103            images = self.main_view.controller.image_ctrl.handle_charger_images_patient(patient_id)
104            self.refresh_images(images)
