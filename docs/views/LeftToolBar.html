335        """Désélectionne tous les boutons de traitement/contraste sauf celui spécifié."""
336        buttons = [self.btn_gaussian, self.btn_low_pass, self.btn_high_pass, self.btn_sobel, self.btn_clahe, self.btn_watershed, self.btn_contrast_slider]
337        for btn in buttons:
